function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); // 1. ถ้าบอร์ดส่งข้อมูลมาบันทึก (มีค่า temp ส่งมา) if (e.parameter.temp != null) { var now = new Date(); var date = Utilities.formatDate(now, "Asia/Bangkok", "dd/MM/yyyy"); var time = Utilities.formatDate(now, "Asia/Bangkok", "HH:mm:ss"); // บันทึกค่าลงแถวที่ 2 sheet.getRange(2, 1).setValue(date); sheet.getRange(2, 2).setValue(time); sheet.getRange(2, 3).setValue(e.parameter.temp); sheet.getRange(2, 4).setValue(e.parameter.volt); sheet.getRange(2, 5).setValue(e.parameter.smoke); sheet.getRange(2, 6).setValue(now.getTime()); // บันทึกเวลา Timestamp ไว้เช็คค้าง return ContentService.createTextOutput("OK"); } // 2. ถ้าหน้าเว็บดึงข้อมูลไปแสดงผล var data = sheet.getRange(2, 1, 1, 6).getValues()[0]; var lastTimestamp = data[5]; // เวลาที่บอร์ดส่งมาล่าสุด var currentTime = new Date().getTime(); // เวลาปัจจุบันของระบบ // คำนวณความต่างเวลา (หน่วยวินาที) var diffInSeconds = (currentTime - lastTimestamp) / 1000; // ถ้านานเกิน 15 วินาที หรือยังไม่มีข้อมูลบันทึก ให้ส่งสถานะ offline if (!lastTimestamp || diffInSeconds > 15) { var json = { status: "offline", temp: 0, volt: 0, smoke: 0, date: "-", time: "-" }; } else { // ถ้าระบบส่งข้อมูลมาปกติ (ไม่เกิน 15 วินาที) var json = { status: "online", date: data[0], time: data[1], temp: data[2], volt: data[3], smoke: data[4] }; } return ContentService .createTextOutput(JSON.stringify(json)) .setMimeType(ContentService.MimeType.JSON); }
